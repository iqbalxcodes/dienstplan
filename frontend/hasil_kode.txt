self.addEventListener("push", (event) => {
    let payload = { title: "Dienstplan", body: "You have a new notification." };
    try { payload = event.data.json(); } catch (err) { console.error("Push payload parse failed:", err); }
    event.waitUntil(self.registration.showNotification(payload.title, {
        body: payload.body, icon: "/favicon.ico", badge: "/favicon.ico", data: { url: payload.url || "/" }
    }));
});

self.addEventListener("notificationclick", (event) => {
    event.notification.close();
    const targetUrl = event.notification.data?.url || "/";
    event.waitUntil(clients.matchAll({ type: "window" }).then((list) => {
        for (const c of list) if (c.url.includes(targetUrl) && "focus" in c) return c.focus();
        if (clients.openWindow) return clients.openWindow(targetUrl);
    }));
});

// One popup for: Assign (new), Edit Shift (manager), Request (manager approves/rejects).
import { useState } from "react";
import { isOff, type RosterForm } from "../../lib/roster";
import type { Membership, Shift, ShiftRequest, ShiftType } from "../../lib/types";

export type PopupSpec =
    | { mode: "assign"; dates: string[] }
    | { mode: "edit"; shift: Shift }
    | { mode: "request"; req: ShiftRequest };

interface Props {
    spec: PopupSpec;
    members: Membership[];
    types: ShiftType[];
    areas: string[];
    roles: string[];
    isManager: boolean;
    meId: string;
    onClose: () => void;
    onSubmit: (f: RosterForm, dates: string[]) => Promise<string | null>;
    onDelete?: () => Promise<string | null>;   // Delete (edit) / Reject (request)
}

function initialOf(spec: PopupSpec, isManager: boolean, meId: string): RosterForm {
    if (spec.mode === "assign")
        return { date: spec.dates[0] ?? "", membershipId: isManager ? "" : meId, code: "", area: "", role: "", start: "", end: "" };
    if (spec.mode === "edit") {
        const s = spec.shift;
        return { date: s.shift_date, membershipId: s.membership_id, code: s.shift_code ?? "", area: s.area_label ?? "", role: s.role_label ?? "", start: s.start_time.slice(0, 5), end: s.end_time.slice(0, 5) };
    }
    const r = spec.req;
    return { date: r.request_date, membershipId: r.membership_id, code: r.shift_code ?? "", area: r.area_label ?? "", role: r.role_label ?? "", start: r.start_time.slice(0, 5), end: r.end_time.slice(0, 5) };
}

const short = (d: string) => `${d.slice(8)}.${d.slice(5, 7)}.`;
const withCurrent = (list: string[], cur: string) => [...new Set([...list, cur])].filter(Boolean);

export function AssignPopup(p: Props) {

    const [initial] = useState(() => initialOf(p.spec, p.isManager, p.meId));
    const [f, setF] = useState(initial);
    const [err, setErr] = useState("");
    const [busy, setBusy] = useState(false);

    const dates = p.spec.mode === "assign" ? p.spec.dates : [];
    const multi = dates.length > 1;
    const ds = multi ? dates : f.date ? [f.date] : [];
    const off = isOff(p.types.find(t => t.code === f.code));
    const set = (patch: Partial<RosterForm>) => setF(x => ({ ...x, ...patch }));

    const missing = [
        ds.length === 0 && "date", !f.membershipId && "employee", !f.code && "shift",
        ...(off ? [] : [!f.area && "area", !f.role && "role", !f.start && "start time", !f.end && "end time"])
    ].filter(Boolean) as string[];
    const valid = missing.length === 0;
    const dirty = JSON.stringify(f) !== JSON.stringify(initial);

    const title = p.spec.mode === "assign" ? "Assign" : p.spec.mode === "edit" ? "Edit Shift" : "Request";
    const submitLabel = p.spec.mode === "assign" ? (p.isManager ? "Save" : "Send request")
        : p.spec.mode === "edit" ? "Save" : (dirty ? "Save" : "Approve");

    function pickShift(code: string) {
        const t = p.types.find(x => x.code === code);
        set({ code, ...(t && !isOff(t) ? { start: t.start ?? f.start, end: t.end ?? f.end } : {}) });
    }

    async function run(fn: () => Promise<string | null>) {
        setBusy(true);
        setErr("");
        const e = await fn();
        setBusy(false);
        if (e) setErr(e); else p.onClose();
    }

    return (
        <div className="plan-modal-backdrop">
            <div className="plan-modal">
                <h3>{title}</h3>

                <label>Date</label>
                {multi
                    ? <input disabled value={dates.map(short).join(", ")} />
                    : <input type="date" value={f.date} onChange={e => set({ date: e.target.value })} />}

                <label>Employee</label>
                <select value={f.membershipId} disabled={!p.isManager} onChange={e => set({ membershipId: e.target.value })}>
                    <option value="">— select —</option>
                    {p.members.map(m => <option key={m.id} value={m.id}>{m.full_name}</option>)}
                </select>

                <label>Shift</label>
                <select value={f.code} onChange={e => pickShift(e.target.value)}>
                    <option value="">— select —</option>
                    {withCurrent(p.types.map(t => t.code), f.code).map(c => {
                        const t = p.types.find(x => x.code === c);
                        return <option key={c} value={c}>{t ? `${c} — ${t.name}` : c}</option>;
                    })}
                </select>

                {!off && (
                    <>
                        <label>Area</label>
                        <select value={f.area} onChange={e => set({ area: e.target.value })}>
                            <option value="">— select —</option>
                            {withCurrent(p.areas, f.area).map(a => <option key={a} value={a}>{a}</option>)}
                        </select>

                        <label>Role</label>
                        <select value={f.role} onChange={e => set({ role: e.target.value })}>
                            <option value="">— select —</option>
                            {withCurrent(p.roles, f.role).map(r => <option key={r} value={r}>{r}</option>)}
                        </select>

                        <label>Start time</label>
                        <input type="time" value={f.start} onChange={e => set({ start: e.target.value })} />
                        <label>End time</label>
                        <input type="time" value={f.end} onChange={e => set({ end: e.target.value })} />
                    </>
                )}

                {!valid && <div className="plan-entry-meta" style={{ marginTop: 12, color: "#c62828" }}>Please fill in all fields (missing: {missing.join(", ")}).</div>}
                {err && <div className="auth-error">{err}</div>}

                <div className="plan-modal-footer">
                    {p.onDelete && (
                        <button className="danger" disabled={busy} onClick={() => {
                            if (p.spec.mode === "edit" && !confirm("Delete this shift?")) return;
                            void run(p.onDelete!);
                        }}>{p.spec.mode === "edit" ? "Delete" : "Reject"}</button>
                    )}
                    <button onClick={p.onClose}>Cancel</button>
                    <button className="primary" disabled={!valid || busy} onClick={() => run(() => p.onSubmit(f, ds))}>{submitLabel}</button>
                </div>
            </div>
        </div>
    );
}

// Google-Calendar-style month grid. Pointer events: same code for mouse and touch.
import { useRef, type PointerEvent as RPointerEvent } from "react";
import { useAuth } from "../../lib/auth";
import { addDays, isSameDay, iso, startOfMonth, startOfToday } from "../../lib/dates";
import type { ShiftFields } from "../../lib/shiftEdit";
import type { Membership, Shift, ShiftChangeRequest } from "../../lib/types";

export type PopupSpec = { shiftId?: string; requestId?: string; date?: string };

const HEAD = ["MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN"];

export function buildWeeks(monthDate: Date): Date[][] {
    const first = startOfMonth(monthDate);
    const lead = (first.getDay() + 6) % 7; // Monday-first
    const inMonth = new Date(monthDate.getFullYear(), monthDate.getMonth() + 1, 0).getDate();
    const weeks = Math.ceil((lead + inMonth) / 7);
    const start = addDays(first, -lead);
    return Array.from({ length: weeks }, (_, w) => Array.from({ length: 7 }, (_, d) => addDays(start, w * 7 + d)));
}

interface Props {
    monthDate: Date;
    members: Membership[];
    shifts: Shift[];
    changes: ShiftChangeRequest[];
    onPopup: (spec: PopupSpec) => void;
    onStep: (months: number) => void;
    save: (shiftId: string, ownerId: string, f: ShiftFields) => void | Promise<void>;
}

export function Month({ monthDate, members, shifts, changes, onPopup, onStep, save }: Props) {

    const { me, isManager } = useAuth();
    const dragging = useRef(false);
    const touch = useRef({ x: 0, y: 0 });
    const weeks = buildWeeks(monthDate);
    const today = startOfToday();
    const nameOf = (id: string) => members.find(m => m.id === id)?.full_name;

    function onDown(e: RPointerEvent) {

        const card = (e.target as HTMLElement).closest(".month-shift-card") as HTMLElement | null;
        if (!card) return;

        const isGhost = card.classList.contains("pending-ghost");
        const owner = card.dataset.membershipId!;
        const canEdit = !isGhost && (isManager || owner === me?.id);
        if (!canEdit && !isGhost) return; // others' shifts: not interactive

        const id = (isGhost ? card.dataset.requestId : card.dataset.shiftId)!;
        const startX = e.clientX, startY = e.clientY;
        let moved = false;
        dragging.current = true;

        let ghost: HTMLElement | null = null;
        const place = (ev: { clientX: number; clientY: number }) => {
            if (!ghost) return;
            ghost.style.left = ev.clientX - ghost.offsetWidth / 2 + "px";
            ghost.style.top = ev.clientY - 16 + "px";
        };

        if (canEdit) {
            ghost = card.cloneNode(true) as HTMLElement;
            ghost.classList.add("rack-drag-ghost");
            ghost.style.position = "fixed";
            ghost.style.width = card.getBoundingClientRect().width + "px";
            ghost.style.pointerEvents = "none";
            document.body.appendChild(ghost);
            place(e);
        }

        const clearTarget = () => document.querySelectorAll(".month-day-cell.drag-target").forEach(el => el.classList.remove("drag-target"));

        const move = (ev: PointerEvent) => {
            if (Math.hypot(ev.clientX - startX, ev.clientY - startY) > 6) moved = true;
            if (!ghost) return;
            place(ev);
            clearTarget();
            document.elementFromPoint(ev.clientX, ev.clientY)?.closest(".month-day-cell")?.classList.add("drag-target");
        };

        const up = (ev: PointerEvent) => {
            document.removeEventListener("pointermove", move);
            document.removeEventListener("pointerup", up);
            dragging.current = false;
            clearTarget();
            ghost?.remove();

            if (!moved) { onPopup(isGhost ? { requestId: id } : { shiftId: id }); return; }
            if (isGhost) return; // pending requests can't be re-dated

            const date = (document.elementFromPoint(ev.clientX, ev.clientY)?.closest(".month-day-cell") as HTMLElement | null)?.dataset.date;
            const shift = shifts.find(s => s.id === id);
            if (!date || !shift || date === shift.shift_date) return;

            void save(id, owner, { shift_date: date, start_time: shift.start_time, end_time: shift.end_time });
        };

        document.addEventListener("pointermove", move);
        document.addEventListener("pointerup", up);
    }

    return (
        <div className="month-view" id="monthView"
            onTouchStart={e => { if (!dragging.current) touch.current = { x: e.touches[0].clientX, y: e.touches[0].clientY }; }}
            onTouchEnd={e => {
                if (dragging.current) return;
                const dy = e.changedTouches[0].clientY - touch.current.y;
                const dx = e.changedTouches[0].clientX - touch.current.x;
                if (Math.abs(dy) > 60 && Math.abs(dy) > Math.abs(dx)) onStep(dy < 0 ? 1 : -1); // swipe up = next month
            }}>

            <div className="month-header-row">{HEAD.map(d => <div key={d} className="month-header-cell">{d}</div>)}</div>

            <div className="month-grid" id="monthGrid" style={{ gridTemplateRows: `repeat(${weeks.length}, 1fr)` }}
                onPointerDown={onDown}
                onClick={e => {
                    const t = e.target as HTMLElement;
                    if (t.closest(".month-shift-card") || !isManager) return;
                    const date = (t.closest(".month-day-cell") as HTMLElement | null)?.dataset.date;
                    if (date) onPopup({ date });
                }}>

                {weeks.flat().map(d => {
                    const key = iso(d);
                    const inMonth = d.getMonth() === monthDate.getMonth() && d.getFullYear() === monthDate.getFullYear();
                    return (
                        <div key={key} className={"month-day-cell" + (inMonth ? "" : " other-month")} data-date={key}>
                            <div className={"month-day-num" + (isSameDay(d, today) ? " is-today" : "")}>{d.getDate()}</div>
                            <div className="month-cards">
                                {shifts.filter(s => s.shift_date === key && nameOf(s.membership_id)).map(s => (
                                    <div key={s.id} className="month-shift-card"
                                        data-shift-id={s.id} data-membership-id={s.membership_id}>
                                        <span className="month-card-name">{nameOf(s.membership_id)}</span>
                                        <span className="month-card-meta">{s.role_label ?? ""} · {s.start_time.slice(0, 5)}–{s.end_time.slice(0, 5)}</span>
                                    </div>
                                ))}
                                {changes.filter(r => r.proposed_shift_date === key && nameOf(r.requested_by_membership_id)).map(r => (
                                    <div key={r.id} className="month-shift-card pending-ghost"
                                        data-request-id={r.id} data-membership-id={r.requested_by_membership_id}>
                                        <span className="month-card-name">{nameOf(r.requested_by_membership_id)}</span>
                                        <span className="month-card-meta">{r.proposed_start_time.slice(0, 5)}–{r.proposed_end_time.slice(0, 5)} · pending</span>
                                    </div>
                                ))}
                            </div>
                        </div>
                    );
                })}
            </div>
        </div>
    );
}

// Gantt-style staff rack: one row per member, shift bars positioned by real clock time.
import type { MouseEvent as RMouseEvent } from "react";
import { useAuth } from "../../lib/auth";
import { useStatus } from "../../lib/status";
import { DOW, isSameDay, iso, MONTHS, parseDate, startOfToday } from "../../lib/dates";
import { getShiftRange, rangeToShiftFields, type MinuteRange } from "../../lib/shifts";
import type { ShiftFields } from "../../lib/shiftEdit";
import { LEAVE_LABELS, type LeaveRequest, type Membership, type Shift } from "../../lib/types";

interface Props {
    days: Date[];
    colW: number;
    members: Membership[];
    shifts: Shift[];
    pending: Record<string, number>;   // membership id -> pending time entries
    leave: LeaveRequest[];
    onShift: (days: number) => void;
    save: (shiftId: string, ownerId: string, f: ShiftFields) => void | Promise<void>;
}

export function Rack({ days, colW, members, shifts, pending, leave, onShift, save }: Props) {

    const { me, isManager } = useAuth();
    const { say } = useStatus();
    const today = startOfToday();
    const width = colW * days.length;
    const rangeStart = getShiftRange(iso(days[0]), "00:00", "00:00").start;
    const rangeEnd = getShiftRange(iso(days[days.length - 1]), "00:00", "00:00").start + 1440;
    const pxPerMin = colW / 1440;

    function onDown(e: RMouseEvent) {

        const target = e.target as HTMLElement;
        const bar = target.closest(".plan-shift-bar") as HTMLElement | null;
        if (!bar || e.button !== 0) return;

        const owner = bar.dataset.membershipId!;
        if (!isManager && owner !== me?.id) return;   // others' shifts are read-only

        e.preventDefault();

        const handle = target.closest(".bar-resize-handle");
        const type = handle ? (handle.classList.contains("left") ? "resize-left" : "resize-right") : "move";
        const range = getShiftRange(bar.dataset.shiftDate!, bar.dataset.startTime!, bar.dataset.endTime!);
        const originLeft = parseFloat(bar.style.left);
        const originWidth = parseFloat(bar.style.width);
        const startX = e.clientX;
        let moved = false;

        const place = (ev: { clientX: number; clientY: number }) => {
            if (ghost) ghost.style.transform = `translate(${ev.clientX - originWidth / 2}px, ${ev.clientY - 16}px)`;
        };

        let ghost: HTMLElement | null = null;
        if (type === "move") {
            ghost = bar.cloneNode(true) as HTMLElement;
            ghost.classList.add("rack-drag-ghost");
            Object.assign(ghost.style, {
                position: "fixed", left: "0px", top: "0px", width: originWidth + "px",
                height: bar.getBoundingClientRect().height + "px", pointerEvents: "none"
            });
            document.body.appendChild(ghost);
            place(e);
        }

        const snapPx = colW / 96; // 15-minute steps

        const move = (ev: MouseEvent) => {
            const dx = ev.clientX - startX;
            if (Math.abs(dx) > 4) moved = true;

            if (type === "move") { place(ev); return; }

            const steps = Math.round(dx / snapPx);
            if (type === "resize-left") {
                const w = originWidth - steps * snapPx;
                if (w < snapPx * 2) return;
                bar.style.left = originLeft + steps * snapPx + "px";
                bar.style.width = w + "px";
            } else {
                const w = originWidth + steps * snapPx;
                if (w < snapPx * 2) return;
                bar.style.width = w + "px";
            }
        };

        const up = (ev: MouseEvent) => {
            document.removeEventListener("mousemove", move);
            document.removeEventListener("mouseup", up);
            ghost?.remove();

            // React won't repaint a bar whose props didn't change: undo the live resize ourselves
            bar.style.left = originLeft + "px";
            bar.style.width = originWidth + "px";

            if (!moved) return;

            const delta = Math.round(((ev.clientX - startX) / pxPerMin) / 15) * 15;
            const next: MinuteRange = { ...range };
            if (type === "resize-left") next.start += delta;
            else if (type === "resize-right") next.end += delta;
            else { next.start += delta; next.end += delta; }

            if (next.end - next.start < 15) { say("Shift must be at least 15 minutes", "error"); return; }

            void save(bar.dataset.shiftId!, owner, rangeToShiftFields(next));
        };

        document.addEventListener("mousemove", move);
        document.addEventListener("mouseup", up);
    }

    return (
        <div className="rack-scroll" id="rackScroll">

            <div id="rackHeaderRow" className="rack-header-row">
                <div className="rack-header-label">
                    <span>STAFF</span>
                    <button className="rack-nav-btn rack-nav-prev" onClick={() => onShift(-1)} title="Previous">{"\u25c0"}</button>
                </div>
                {days.map(d => (
                    <div key={iso(d)} style={{ width: colW }}
                        className={"rack-header-cell" + (isSameDay(d, today) ? " is-today" : "") + ([0, 6].includes(d.getDay()) ? " is-weekend" : "")}>
                        <div className="rack-header-dow">{DOW[d.getDay()]}</div>
                        <div className="rack-header-date">{String(d.getDate()).padStart(2, "0")} {MONTHS[d.getMonth()]}</div>
                    </div>
                ))}
                <div className="rack-header-end">
                    <button className="rack-nav-btn" onClick={() => onShift(1)} title="Next">{"\u25b6"}</button>
                </div>
            </div>

            <div id="rackBody" onMouseDown={onDown}>
                {members.length === 0 && <div style={{ padding: 16, color: "#888" }}>No staff found</div>}

                {members.map(m => (
                    <div key={m.id} className="rack-row plan-row">
                        <div className="rack-room-label">
                            {pending[m.id] > 0 && <span className="rack-conflict-flag" title={`${pending[m.id]} pending approval`}>{"\u26a0"}</span>}
                            <span className="rack-room-number">{m.full_name}</span>
                            {m.id === me?.id && <span className="rack-status-dot status-clean" title="You">{"\u25cf"}</span>}
                        </div>

                        <div className="rack-timeline" style={{ width }}>
                            {days.map(d => (
                                <div key={iso(d)} className={"rack-cell" + ([0, 6].includes(d.getDay()) ? " is-weekend" : "")} style={{ width: colW }} />
                            ))}

                            <div className="rack-bars-layer" style={{ width }}>
                                {shifts.filter(s => s.membership_id === m.id).map(s => {
                                    const r = getShiftRange(s.shift_date, s.start_time, s.end_time);
                                    const a = Math.max(r.start, rangeStart), b = Math.min(r.end, rangeEnd);
                                    if (b <= a) return null;

                                    const cutL = r.start < rangeStart, cutR = r.end > rangeEnd;
                                    const canDrag = isManager || m.id === me?.id;
                                    const meta = [s.role_label, `${s.start_time.slice(0, 5)}\u2013${s.end_time.slice(0, 5)}`, s.is_night_shift ? "Nachtdienst" : null]
                                        .filter(Boolean).join(" \u00b7 ");

                                    return (
                                        <div key={s.id}
                                            className={"reservation-bar plan-shift-bar" + (cutL ? " cut-left" : "") + (cutR ? " cut-right" : "")}
                                            style={{ left: (a - rangeStart) * pxPerMin + 3, width: (b - a) * pxPerMin - 6, top: 2, height: "calc(100% - 4px)" }}
                                            title={`${m.full_name} \u00b7 ${meta}`}
                                            data-shift-id={s.id} data-membership-id={m.id} data-shift-date={s.shift_date}
                                            data-start-time={s.start_time} data-end-time={s.end_time}>
                                            {canDrag && !cutL && <div className="bar-resize-handle left" />}
                                            <span className="bar-guest">{m.full_name}</span>
                                            <span className="bar-meta">{meta}</span>
                                            {canDrag && !cutR && <div className="bar-resize-handle right" />}
                                        </div>
                                    );
                                })}

import { DEFAULT_ROLES, type Membership, type Shift, type ShiftChangeRequest } from "../../lib/types";
import type { PopupSpec } from "./Month";

interface Props {
    spec: PopupSpec;
    members: Membership[];
    shifts: Shift[];
    changes: ShiftChangeRequest[];
    onClose: () => void;
    onSave: (shift: Shift, f: { start: string; end: string; role: string }) => void;
    onCreate: (membershipId: string, date: string, start: string, end: string, role: string) => void;

export function ShiftPopup({ spec, members, shifts, changes, onClose, onSave, onCreate }: Props) {

    const { org, isManager } = useAuth();
    const roles = org?.settings?.role_labels ?? DEFAULT_ROLES;

    const shift = shifts.find(s => s.id === spec.shiftId);
    const req = changes.find(r => r.id === spec.requestId);

    const [employee, setEmployee] = useState(members[0]?.id ?? "");
    const [role, setRole] = useState(shift?.role_label ?? roles[0]);
    const [start, setStart] = useState((shift?.start_time ?? req?.proposed_start_time ?? "09:00").slice(0, 5));
    const [end, setEnd] = useState((shift?.end_time ?? req?.proposed_end_time ?? "17:00").slice(0, 5));

    const readOnly = !!req;
    const adding = !shift && !req;
    const who = members.find(m => m.id === (shift?.membership_id ?? req?.requested_by_membership_id))?.full_name ?? "";
    const title = readOnly ? "Pending Change" : adding ? "Add Shift" : "Edit Shift";

    function submit() {
        if (!start || !end) return;
        if (shift) onSave(shift, { start, end, role });
        else if (spec.date && employee) onCreate(employee, spec.date, start + ":00", end + ":00", role);
        onClose();
    }

    return (
        <Modal title={title} onClose={onClose} onSubmit={readOnly ? undefined : submit}>
            <div className="plan-entry-meta" style={{ marginBottom: 8 }}>{who}</div>

            {adding && (
                <>
                    <label>Employee</label>
                    <select value={employee} onChange={e => setEmployee(e.target.value)}>
                        {members.map(m => <option key={m.id} value={m.id}>{m.full_name}</option>)}
                    </select>
                </>
            )}

            <label>Role</label>
            {/* role isn't part of a change request, so only managers can change it */}
            <select value={role} disabled={readOnly || !isManager} onChange={e => setRole(e.target.value)}>
                {[...new Set([...roles, role])].map(r => <option key={r} value={r}>{r}</option>)}
            </select>

            <label>Start time</label>
            <input type="time" value={start} disabled={readOnly} onChange={e => setStart(e.target.value)} />
            <label>End time</label>
            <input type="time" value={end} disabled={readOnly} onChange={e => setEnd(e.target.value)} />

            {readOnly && <div className="plan-entry-meta" style={{ marginTop: 8, color: "#e65100" }}>Waiting for manager approval.</div>}
            {shift && !isManager && <div className="plan-entry-meta" style={{ marginTop: 8, color: "#e65100" }}>Your change will be sent for manager approval.</div>}

/** Button + popover. Closes on outside click or when an item is clicked. */
function Menu({ button, className, active, align = "left", children }:
    { button: ReactNode; className: string; active?: boolean; align?: "left" | "right"; children: ReactNode }) {

    const [open, setOpen] = useState(false);
    const ref = useRef<HTMLDivElement>(null);

    useEffect(() => {
        if (!open) return;
        const close = (e: MouseEvent) => { if (!ref.current?.contains(e.target as Node)) setOpen(false); };
        document.addEventListener("mousedown", close);
        return () => document.removeEventListener("mousedown", close);
    }, [open]);

    return (
        <div className="hdr-menu" ref={ref}>
            <button className={className + (open || active ? " active" : "")} onClick={() => setOpen(o => !o)}>{button}</button>
            {open && <div className={`hdr-pop ${align}`} onClick={() => setOpen(false)}>{children}</div>}
        </div>
    );
}

function OnboardCard({ onBack }: { onBack: () => void }) {

    const { user, reload } = useAuth();
    // signed in (email confirmed) but no membership yet -> resume at step 2
    const pendingName = localStorage.getItem(ONBOARD_PENDING_KEY);
    const [step, setStep] = useState<1 | 2>(user && pendingName ? 2 : 1);
    const [name, setName] = useState(pendingName ?? "");
    const [email, setEmail] = useState("");
    const [pass, setPass] = useState("");
    const [orgName, setOrgName] = useState("");
    const [crew, setCrew] = useState<CrewRow[]>([{ role: "employee", name: "", email: "" }]);
    const [err, setErr] = useState("");

    const setRow = (i: number, patch: Partial<CrewRow>) => setCrew(c => c.map((r, j) => (j === i ? { ...r, ...patch } : r)));

    async function next() {
        setErr("");
        if (!name.trim() || !email.trim() || pass.length < 6) { setErr("Please fill in all fields (password min. 6 characters)."); return; }

        const { data, error } = await supabase.auth.signUp({
            email: email.trim(), password: pass, options: { emailRedirectTo: window.location.origin }
        });
        if (error) { setErr(error.message); return; }

        if (!data.session) {
            localStorage.setItem(ONBOARD_PENDING_KEY, name.trim());
            setErr("Account created — check your email to confirm it. You'll come back here automatically.");
            return;
        }
        setStep(2);
    }

    async function finish() {
        setErr("");
        if (!orgName.trim()) { setErr("Organization name is required."); return; }

        const { data: { user: u } } = await supabase.auth.getUser();
        if (!u) { setErr("Session expired — please log in again."); return; }

        const { data: org, error: orgErr } = await supabase
            .from("organizations").insert({ name: orgName.trim(), slug: slugify(orgName.trim()) }).select().single();
        if (orgErr || !org) { setErr(orgErr?.message ?? "Failed to create organization."); return; }

        const { error: memErr } = await supabase.from("memberships").insert({
            organization_id: org.id, user_id: u.id, role: "admin", full_name: name.trim(), email: u.email
        });
        if (memErr) { setErr(memErr.message); return; }

        // extra crew: no login yet (user_id null) — they get a recovery link later.

import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from "react";

const REMEMBER_KEY = "dienstplan_remember";
const TAB_ALIVE_KEY = "dienstplan_tab_alive";

type Row = Membership & { organization: Organization };

interface Ctx {
    ready: boolean;
    user: User | null;               // signed in (may still lack a membership: onboarding)
    org: Organization | null;
    me: Membership | null;
    choices: Row[] | null;           // set when user belongs to >1 org and must pick
    isManager: boolean;              // manager OR admin
    isAdmin: boolean;
    reload: () => Promise<void>;
    choose: (membershipId: string) => void;
    logout: () => Promise<void>;
    patchOrg: (patch: Partial<Organization>) => void;
}

const AuthCtx = createContext<Ctx>(null!);
export const useAuth = () => useContext(AuthCtx);

export function AuthProvider({ children }: { children: ReactNode }) {

    const [ready, setReady] = useState(false);
    const [user, setUser] = useState<User | null>(null);
    const [org, setOrg] = useState<Organization | null>(null);
    const [me, setMe] = useState<Membership | null>(null);
    const [choices, setChoices] = useState<Row[] | null>(null);

    const apply = (row: Row | null) => {
        setMe(row ? (({ organization: _o, ...m }) => m as Membership)(row) : null);
        setOrg(row?.organization ?? null);
        setChoices(null);
    };

    const reload = useCallback(async () => {

        // "Don't remember me": no tab sentinel = all windows were closed -> drop session
        if (localStorage.getItem(REMEMBER_KEY) === "0") {
            if (!sessionStorage.getItem(TAB_ALIVE_KEY)) await supabase.auth.signOut();
            sessionStorage.setItem(TAB_ALIVE_KEY, "1");
        }

        const { data: { session } } = await supabase.auth.getSession();
        setUser(session?.user ?? null);

        if (!session) { apply(null); setReady(true); return; }

        const { data: mine, error } = await supabase
            .from("memberships")
            .select("*, organization:organizations(*)")
            .eq("user_id", session.user.id)
            .eq("active", true);

        if (error || !mine || mine.length === 0) {
            if (error) console.error("Failed to load memberships", error);
            apply(null);
        } else if (mine.length === 1) {
            apply(mine[0] as Row);
        } else {
            apply(null);
            setChoices(mine as Row[]);
        }

        setReady(true);
    }, []);

    useEffect(() => { void reload(); }, [reload]);

    const choose = (id: string) => apply(choices?.find(c => c.id === id) ?? null);

    const logout = async () => {
        try { await supabase.auth.signOut(); }
        catch (err) { console.error("signOut failed, clearing local state anyway:", err); }
        setUser(null);
        apply(null);
    };

    const patchOrg = (patch: Partial<Organization>) => setOrg(o => (o ? { ...o, ...patch } : o));

    const value = useMemo<Ctx>(() => ({
        ready, user, org, me, choices,
        isManager: me?.role === "manager" || me?.role === "admin",
        isAdmin: me?.role === "admin",
        reload, choose, logout, patchOrg
    // eslint-disable-next-line react-hooks/exhaustive-deps
    }), [ready, user, org, me, choices, reload]);

    return <AuthCtx.Provider value={value}>{children}</AuthCtx.Provider>;
}

    orgId: string, membershipId: string, timeEntryId: string, message: string, files: File[]

        .from("complaints")
        .insert({ organization_id: orgId, time_entry_id: timeEntryId, membership_id: membershipId, message, status: "open" })
        .select()
        .single();

    if (error || !complaint) return error?.message ?? "Failed to file complaint";

    for (const file of files) {

        const { error: evErr } = await supabase
            .from("complaint_evidence")
            .insert({ complaint_id: complaint.id, file_path: path, file_name: file.name });
        if (evErr) console.error(evErr);
    }

    return null;
}

// Soll / Ist / Überstunden from APPROVED time_entries only.
import { supabase } from "./supabase";
import type { Membership, HoursSummary } from "./types";

const round2 = (n: number) => Math.round(n * 100) / 100;

export function sollHours(m: Membership, start: Date, end: Date): number {
    const days = Math.max(1, Math.round((end.getTime() - start.getTime()) / 86400000) + 1);
    return (m.weekly_target_hours / 7) * days;
}

export async function computeHours(orgId: string, m: Membership, start: Date, end: Date): Promise<HoursSummary> {

    const soll = sollHours(m, start, end);

    const { data, error } = await supabase
        .from("time_entries")
        .select("clock_in, clock_out, shifts:shift_id(break_minutes)")
        .eq("organization_id", orgId)
        .eq("membership_id", m.id)
        .eq("status", "approved")
        .gte("clock_in", start.toISOString())
        .lt("clock_in", new Date(end.getTime() + 86400000).toISOString());

    if (error || !data) {
        console.error(error);
        return { sollHours: round2(soll), istHours: 0, ueberstunden: round2(-soll) };
    }

    const ist = (data as any[]).reduce((sum, e) => {
        if (!e.clock_in || !e.clock_out) return sum;
        const raw = (Date.parse(e.clock_out) - Date.parse(e.clock_in)) / 60000;
        return sum + Math.max(0, raw - (e.shifts?.break_minutes ?? 0)) / 60;
    }, 0);

    return { sollHours: round2(soll), istHours: round2(ist), ueberstunden: round2(ist - soll) };
}

export function formatHours(h: number): string {
    const abs = Math.abs(h);
    const hh = Math.floor(abs);
    const mm = Math.round((abs - hh) * 60);
    return `${h < 0 ? "-" : ""}${hh}h ${String(mm).padStart(2, "0")}m`;
}

// Web Push: registers /sw.js and stores the subscription in Supabase so the
// server-side cron (Edge Function) can notify even when no tab is open.
import { supabase } from "./supabase";

const VAPID_PUBLIC_KEY = import.meta.env.VITE_VAPID_PUBLIC_KEY as string;

export const isPushSupported = () => "serviceWorker" in navigator && "PushManager" in window;

function keyToBytes(base64: string): Uint8Array<ArrayBuffer> {
    const b64 = (base64 + "=".repeat((4 - base64.length % 4) % 4)).replace(/-/g, "+").replace(/_/g, "/");
    const raw = atob(b64);
    return Uint8Array.from(raw, c => c.charCodeAt(0));
}

export async function enablePush(orgId: string, membershipId: string): Promise<boolean> {

    if (!isPushSupported()) return false;
    if (await Notification.requestPermission() !== "granted") return false;

    try {
        const reg = await navigator.serviceWorker.register("/sw.js");
        await navigator.serviceWorker.ready;

        const sub = (await reg.pushManager.getSubscription())
            ?? await reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: keyToBytes(VAPID_PUBLIC_KEY) });

        const json = sub.toJSON();

        const { error } = await supabase.from("push_subscriptions").upsert({
            membership_id: membershipId,
            organization_id: orgId,
            endpoint: json.endpoint!,
            p256dh: json.keys!.p256dh,
            auth: json.keys!.auth
        }, { onConflict: "endpoint" });

        if (error) { console.error("Failed to store push subscription:", error); return false; }
        return true;
    } catch (err) {
        console.error("Push subscription failed:", err);
        return false;
    }
}

export async function disablePush(): Promise<void> {

    if (!isPushSupported()) return;

    try {
        const sub = await (await navigator.serviceWorker.getRegistration())?.pushManager.getSubscription();
        if (sub) {
            const endpoint = sub.endpoint;
            await sub.unsubscribe();
            await supabase.from("push_subscriptions").delete().eq("endpoint", endpoint);
        }
    } catch (err) {
        console.error("Failed to disable push:", err);
    }
}

// Shared shift write paths for rack + month view.
// Manager: writes `shifts` directly (after conflict check).

import { supabase } from "./supabase";

export interface ShiftFields { shift_date: string; start_time: string; end_time: string; role_label?: string; }

export function useShiftEdit(reload: () => void | Promise<void>) {

    const { say } = useStatus();

    async function apply(shiftId: string, membershipId: string, f: ShiftFields) {

        const { conflicts, error: cErr } = await findShiftConflicts(
            org!.id, membershipId, f.shift_date, f.start_time, f.end_time, shiftId);

        if (cErr) { say("Failed to check for conflicts", "error"); return reload(); }
        if (conflicts.length) { say("This staff member already has an overlapping shift", "error"); return reload(); }

            updated_by: user?.id ?? null
        }).eq("id", shiftId);

        if (error) { console.error(error); say("Failed to update shift", "error"); }
        else say("Shift updated", "success");
        await reload();
    }

    async function propose(shiftId: string, f: ShiftFields) {

        // shift_change_requests has no role column: role edits are manager-only
        const { error } = await supabase.from("shift_change_requests").insert({
            organization_id: org!.id,
            shift_id: shiftId,
            requested_by_membership_id: me!.id,
            proposed_shift_date: f.shift_date,
            proposed_start_time: f.start_time,
            proposed_end_time: f.end_time
        });

        if (error) { console.error(error); say("Failed to submit change request", "error"); }
        else say("Change requested — waiting for manager approval", "success");
        await reload();
    }

    const save = (shiftId: string, ownerId: string, f: ShiftFields) =>
        isManager ? apply(shiftId, ownerId, f) : propose(shiftId, f);

    async function create(membershipId: string, date: string, start: string, end: string, role: string) {

        if (!isManager) return;

        const { conflicts, error: cErr } = await findShiftConflicts(org!.id, membershipId, date, start, end);
        if (cErr) { say("Failed to check for conflicts", "error"); return; }
        if (conflicts.length) { say("This staff member already has an overlapping shift", "error"); return; }

            start_time: start, end_time: end, role_label: role, updated_by: user?.id ?? null
        });

        if (error) { console.error(error); say("Failed to create shift", "error"); return; }
        say("Shift added", "success");
        await reload();
    }

    return { save, create };

// Shift time math + double-booking check.
// Ranges are in minutes since epoch, anchored at UTC midnight so every
// day is exactly 1440 min (DST-proof). Night shifts (end <= start) roll
// onto the next day.
import { supabase } from "./supabase";
import type { Shift } from "./types";

export interface MinuteRange { start: number; end: number; }

const DAY = 1440;

function dateToMinutes(dateStr: string): number {
    const [y, m, d] = dateStr.split("-").map(Number);
    return Date.UTC(y, m - 1, d) / 60000;
}

const minutesOfDay = (t: string) => {
    const [h, m] = t.split(":").map(Number);
    return h * 60 + m;
};

export function getShiftRange(shiftDate: string, startTime: string, endTime: string): MinuteRange {
    const day = dateToMinutes(shiftDate);
    const start = day + minutesOfDay(startTime);
    let end = day + minutesOfDay(endTime);
    if (end <= start) end += DAY;
    return { start, end };
}

const overlap = (a: MinuteRange, b: MinuteRange) => a.start < b.end && b.start < a.end;

const dateFromMinutes = (min: number) => new Date(min * 60000).toISOString().slice(0, 10);

const toTime = (total: number) =>
    `${String(Math.floor(total / 60) % 24).padStart(2, "0")}:${String(total % 60).padStart(2, "0")}:00`;

/** Absolute range back to shift_date/start_time/end_time. */
export function rangeToShiftFields(range: MinuteRange) {
    const dayStart = Math.floor(range.start / DAY) * DAY;
    return {
        shift_date: dateFromMinutes(dayStart),
        start_time: toTime(range.start - dayStart),
        end_time: toTime((range.end - dayStart) % DAY)
    };
}

/** Other shifts of this member that overlap the proposed one (±1 day window catches night shifts). */
export async function findShiftConflicts(
    orgId: string, membershipId: string, date: string, start: string, end: string, excludeId?: string
): Promise<{ conflicts: Shift[]; error: unknown }> {

    const day = dateToMinutes(date);

    let q = supabase.from("shifts").select("*")
        .eq("organization_id", orgId)
        .eq("membership_id", membershipId)
        .gte("shift_date", dateFromMinutes(day - DAY))
        .lte("shift_date", dateFromMinutes(day + DAY));
    if (excludeId) q = q.neq("id", excludeId);

    const { data, error } = await q;
    if (error || !data) return { conflicts: [], error };

    const proposed = getShiftRange(date, start, end);
    const conflicts = (data as Shift[]).filter(s =>
        overlap(proposed, getShiftRange(s.shift_date, s.start_time, s.end_time)));

    return { conflicts, error: null };
}

function parseMapsInput(raw: string): [number, number] | null {
    const s = decodeURIComponent(raw.trim());
    const m = s.match(/@(-?\d{1,3}\.\d+),(-?\d{1,3}\.\d+)/)
        ?? s.match(/[?&](?:q|ll|query)=(-?\d{1,3}\.\d+)(?:%2C|,)+(-?\d{1,3}\.\d+)/i)
        ?? s.match(/^(-?\d{1,3}\.\d+)\s*,\s*(-?\d{1,3}\.\d+)$/);
    return m ? [parseFloat(m[1]), parseFloat(m[2])] : null;
}

type LL = [number, number];

function haversine(a: LL, b: LL): number {
    const R = 6371000, rad = Math.PI / 180;
    const dLat = (b[0] - a[0]) * rad, dLon = (b[1] - a[1]) * rad;
    const h = Math.sin(dLat / 2) ** 2 + Math.cos(a[0] * rad) * Math.cos(b[0] * rad) * Math.sin(dLon / 2) ** 2;
    return 2 * R * Math.asin(Math.sqrt(h));
}

        };
    }, []);

        if (wp) {

        }

        if (pos) {

            else userPin.current.setLatLng(pos);

                m.fitBounds(L.latLngBounds([wp, pos]).pad(0.4), { animate: true });
                lastFit.current = pos;

        } else if (wp) {
            m.setView(wp, 15);
        }
    }, [wp, pos]);

    return <div ref={el} id="map" className="dsh-map" />;

function Crew() {

    const { org } = useAuth();
    const log = useLog();
    const [list, setList] = useState<Membership[]>([]);
    const [add, setAdd] = useState({ name: "", email: "", role: "employee", hours: "" });

    const load = useCallback(async () => {
        const { data } = await supabase.from("memberships").select("*").eq("organization_id", org!.id).order("full_name");
        setList((data ?? []) as Membership[]);
    }, [org]);

    useEffect(() => { void load(); }, [load]);

    async function addMember() {
        const fullName = add.name.trim();
        const email = add.email.trim().toLowerCase();
        const hours = Number(add.hours) || 40;
        if (!fullName || !email) { alert("Full name and email are required."); return; }

        const { error } = await supabase.from("memberships").insert({
            organization_id: org!.id, user_id: null, email, role: add.role, full_name: fullName, weekly_target_hours: hours
        });
        if (error) { alert(error.message); return; }

        await log("crew.add", `Crew added: ${fullName}`, { email, role: add.role, weekly_hours: hours });
        setAdd({ ...add, name: "", email: "" });
        void load();
    }

    async function saveEdit() {
        if (!edit) return;
        const before = list.find(m => m.id === edit.id)!;
        const patch = {
            full_name: edit.full_name.trim(),
            weekly_target_hours: Number(edit.weekly_target_hours) || 40,
            role: edit.role,
            email: (edit.email ?? "").trim().toLowerCase()
        };
        if (!patch.full_name || !patch.email) { alert("Name and email are required."); return; }

        setEdit(null);
        const { error } = await supabase.from("memberships").update(patch).eq("id", edit.id);
        if (error) { alert(error.message); void load(); return; }

        // role changes are the most security-sensitive action: logged separately
        if (before.role !== patch.role) {
            await log("crew.role_changed", `${patch.full_name}: ${before.role} \u2192 ${patch.role}`,
                { member_email: patch.email, before_role: before.role, after_role: patch.role }, "membership", edit.id);
        } else {
            await log("crew.update", `Crew updated: ${patch.full_name}`, { after: patch }, "membership", edit.id);
        }
        void load();
    }

    async function toggleActive(m: Membership, activate: boolean) {
        // client-side mirror of the "keep at least one active manager/admin" rule
        if (m.role !== "employee" && !activate) {
            const leaders = list.filter(x => x.active && x.role !== "employee");
            if (leaders.length <= 1) { alert("Organization must keep at least one active manager/admin."); return; }
        }

        const { error } = await supabase.from("memberships").update({ active: activate }).eq("id", m.id);
        if (error) { alert(error.message); void load(); return; }

        await log(activate ? "crew.reactivate" : "crew.deactivate",
            `${m.full_name} ${activate ? "reactivated" : "deactivated"}`, {}, "membership", m.id);
        void load();
    }

    async function resendRecovery(m: Membership) {
        if (!m.email) { alert("This member has no email yet \u2014 add one first."); return; }

        const { error } = await supabase.auth.resetPasswordForEmail(m.email, { redirectTo: `${window.location.origin}/reset` });
        if (error) { alert(error.message); return; }

        // recorded server-side (works even when the target has no session)
        await supabase.rpc("log_recovery_sent", { p_email: m.email });
        alert(`Recovery email sent to ${m.email}.`);
    }

    return (
        <>

                <input type="text" placeholder="Full name" value={add.name} onChange={e => setAdd({ ...add, name: e.target.value })} />
                <input type="email" placeholder="Email (for login later)" value={add.email} onChange={e => setAdd({ ...add, email: e.target.value })} />
                <select value={add.role} onChange={e => setAdd({ ...add, role: e.target.value })}>
                    <option value="employee">Crew</option>
                    <option value="manager">Chief/Bos</option>
                </select>
                <input type="number" placeholder="Weekly hours (e.g. 40)" value={add.hours} onChange={e => setAdd({ ...add, hours: e.target.value })} />
                <button onClick={addMember}>Add</button>
            </div>

            </Section>

            <Section title="Crew Members">
            <div id="crewList">
                {list.map(m => (
                    <div key={m.id} className={"plan-entry-card" + (m.active ? "" : " status-rejected")}>
                        <div>
                            <strong>{m.full_name}</strong>{" "}
                            <span title={m.user_id ? "Has login" : "Login not set up yet"}>{m.user_id ? "\ud83d\udfe2" : "\ud83d\udfe1"}</span>{" "}
                            <span className="status-badge">{m.role}</span>{" "}
                            {!m.active && <span className="status-badge status-rejected">inactive</span>}
                            <div className="plan-entry-meta">{m.email ?? "no email"} · {m.weekly_target_hours}h/week</div>
                        </div>
                        <div className="plan-entry-actions">
                            <button onClick={() => setEdit({ ...m })}>Edit</button>
                            <button disabled={!(m.email && m.active)} onClick={() => resendRecovery(m)}>Resend Recovery</button>
                            <button onClick={() => toggleActive(m, !m.active)}>{m.active ? "Deactivate" : "Reactivate"}</button>
                        </div>
                    </div>
                ))}
            </div>
            </Section>

            {edit && (
                <Modal title="Edit Crew Member" onClose={() => setEdit(null)} onSubmit={saveEdit}>
                    <label>Full name</label>
                    <input type="text" value={edit.full_name} onChange={e => setEdit({ ...edit, full_name: e.target.value })} />
                    <label>Weekly target hours</label>
                    <input type="number" value={edit.weekly_target_hours} onChange={e => setEdit({ ...edit, weekly_target_hours: e.target.value as any })} />

        </>
    );
}

import { useEffect, useState, type FormEvent } from "react";
import { useNavigate } from "react-router-dom";
import { supabase } from "../lib/supabase";

export default function ResetPassword() {

    const nav = useNavigate();
    const hasToken = window.location.hash.includes("access_token") || window.location.search.includes("code=");
    const [state, setState] = useState<"loading" | "invalid" | "form" | "done">(hasToken ? "loading" : "invalid");
    const [p1, setP1] = useState("");
    const [p2, setP2] = useState("");
    const [err, setErr] = useState("");
    const [busy, setBusy] = useState(false);

    // supabase-js (detectSessionInUrl) exchanges the link token for a session itself
    useEffect(() => {
        if (!hasToken) return;
        const { data: { subscription } } = supabase.auth.onAuthStateChange((ev, session) => {
            if (ev === "PASSWORD_RECOVERY" || (ev === "SIGNED_IN" && session)) setState("form");
        });
        supabase.auth.getSession().then(({ data }) => { if (data.session) setState("form"); });
        const timeout = setTimeout(() => setState(s => (s === "loading" ? "invalid" : s)), 8000);
        return () => { subscription.unsubscribe(); clearTimeout(timeout); };
    }, [hasToken]);

    async function submit(e: FormEvent) {
        e.preventDefault();
        setErr("");
        if (p1.length < 6) { setErr("Password must be at least 6 characters."); return; }
        if (p1 !== p2) { setErr("Passwords do not match."); return; }

        setBusy(true);
        const { error } = await supabase.auth.updateUser({ password: p1 });
        if (error) { setErr(error.message); setBusy(false); return; }

        setState("done");
        setTimeout(() => nav("/", { replace: true }), 2000);
    }

    return (
        <div className="reset-wrap">
            <div className="plan-modal auth-card reset-card">
                <h3>{state === "form" ? "Set New Password" : state === "done" ? "Done!" : "Reset Password"}</h3>

                {state === "loading" && <div className="reset-loading">Verifying your link…</div>}
                {state === "invalid" && <div className="reset-loading">Link expired or invalid. Please request a new recovery email.</div>}

                {state === "form" && (
                    <form onSubmit={submit}>
                        <label htmlFor="newPassword">New password</label>
                        <input id="newPassword" type="password" placeholder="At least 6 characters" autoComplete="new-password"
                            value={p1} onChange={e => setP1(e.target.value)} />
                        <label htmlFor="confirmPassword">Confirm password</label>
                        <input id="confirmPassword" type="password" placeholder="Repeat your new password" autoComplete="new-password"
                            value={p2} onChange={e => setP2(e.target.value)} />
                        <button type="submit" disabled={busy}>{busy ? "Updating…" : "Update Password"}</button>
                        {err && <div className="reset-msg err">{err}</div>}
                    </form>
                )}

                {state === "done" && <div className="reset-msg ok">Password updated. Redirecting to login…</div>}
            </div>
        </div>
    );
}

            supabase.from("shifts").select("*").eq("organization_id", org!.id).gte("shift_date", from).lte("shift_date", to),
            reqQ,                                                  // table may not exist yet (migration 002) -> empty
            supabase.from("roster_notes").select("*").eq("organization_id", org!.id)
        ]);
        setData({
            members: (m.data ?? []) as Membership[],
            shifts: ((s.data ?? []) as Shift[]).filter(x => isManager || x.status === "published"),   // drafts: managers only until published
            requests: (r.data ?? []) as ShiftRequest[], notes: (n.data ?? []) as RosterNote[]
        });
    }, [org, me, isManager, weeks]);

    useEffect(() => { void load(); }, [load]);

    const byDate = useMemo(() => {
        const map = new Map<string, Entry[]>();
        const add = (d: string, e: Entry) => { if (e.member) map.set(d, [...(map.get(d) ?? []), e]); };
        const member = (id: string) => data.members.find(x => x.id === id);
        data.shifts.forEach(raw => {
            const s = viewOf(raw);
            add(s.shift_date, { key: s.id, code: s.shift_code ?? null, member: member(s.membership_id), shift: s, dirty: raw.status === "draft" || !!raw.draft, del: !!raw.draft?._delete });
        });
        data.requests.forEach(r => add(r.request_date, { key: r.id, code: r.shift_code, member: member(r.membership_id), req: r }));
        const rank = (c: string | null) => { const i = types.findIndex(t => t.code === c); return i < 0 ? 99 : i; };
        map.forEach(l => l.sort((a, b) => rank(a.code) - rank(b.code) || lastName(a.member).localeCompare(lastName(b.member))));
        return map;
    }, [data, types]);

    // "Employee works 6 days non stop!"
    const warnings = useMemo(() => {
        const out: string[] = [];
        const days = new Map<string, Set<string>>();
        data.shifts.forEach(raw => {
            const s = viewOf(raw);
            if (raw.draft?._delete || isOff(typeOf(s.shift_code))) return;
            days.set(s.membership_id, (days.get(s.membership_id) ?? new Set()).add(s.shift_date));
        });
        days.forEach((set, id) => {
            let run = 0, prev = 0;
            [...set].sort().forEach(d => {
                const t = parseDate(d).getTime();
                run = t - prev === 86400000 ? run + 1 : 1;
                prev = t;
                if (run === 6) out.push(`${lastName(data.members.find(m => m.id === id))} works 6 days non stop!`);
            });
        });
        return out;
    }, [data]); // eslint-disable-line react-hooks/exhaustive-deps

    useEffect(() => {
        setNotice(warnings.length ? warnings.slice(0, 3).join("  ·  ") + (warnings.length > 3 ? `  ·  +${warnings.length - 3}` : "") : null);
        return () => setNotice(null);
    }, [warnings, setNotice]);

    const drafts = data.shifts.filter(x => x.status === "draft" || !!x.draft);   // unpublished changes

    const pushUndo = (fn: () => Promise<unknown>) => setUndo(u => [...u, fn]);

    async function doUndo() {
        const fn = undo[undo.length - 1];
        if (!fn) return;
        setUndo(u => u.slice(0, -1));
        await fn();
        await load();
        say("Undone", "info");
    }

    async function publish() {
        if (!drafts.length) { say("Nothing to publish", "info"); return; }
        if (!confirm(`Publish ${drafts.length} change(s)? Employees will see them. This is final and cannot be undone.`)) return;
        const errs = await Promise.all(drafts.map(async x => {
            if (x.draft?._delete) return (await supabase.from("shifts").delete().eq("id", x.id)).error;
            const { _delete, ...fields } = x.draft ?? {};
            return updateShift(x.id, { ...fields, status: "published", draft: null });
        }));
        const bad = errs.find(Boolean);
        if (bad) { say(bad.message, "error"); await load(); return; }
        setUndo([]);   // published = final: nothing before this point can be undone anymore
        say(`Published ${drafts.length} change(s)`, "success");
        await load();
    }

    function toggle(d: string) {
        // employees can only pick days where they have nothing yet
        if (!isManager && (byDate.get(d) ?? []).some(e => e.member?.id === me!.id)) {
            say("You already have an entry on that day", "info");
            return;
        }
        setSelected(s => { const n = new Set(s); if (!n.delete(d)) n.add(d); return n; });
    }

    // ---------- writes (return an error string, or null on success) ----------

    function rowOf(f: RosterForm, date: string) {
        const off = isOff(typeOf(f.code));
        return {
            organization_id: org!.id, membership_id: f.membershipId, shift_date: date,
            start_time: off ? OFF_START : f.start + ":00", end_time: off ? OFF_END : f.end + ":00",
            shift_code: f.code, area_label: off ? null : f.area || null, role_label: off ? null : f.role || null
        };
    }

    async function overlap(f: RosterForm, dates: string[], exclude?: string): Promise<string | null> {
        const r = rowOf(f, dates[0]);
        for (const d of dates) {
            const { conflicts, error } = await findShiftConflicts(org!.id, f.membershipId, d, r.start_time, r.end_time, exclude);
            if (error) return "Failed to check for conflicts";
            if (conflicts.length) return `Overlaps another shift on ${d.slice(8)}.${d.slice(5, 7)}.`;
        }
        return null;
    }

    async function submit(spec: PopupSpec, f: RosterForm, dates: string[]): Promise<string | null> {

        if (spec.mode === "assign" && !isManager) {                       // employee -> request
            const off = isOff(typeOf(f.code));
            const rows = dates.map(d => ({
                organization_id: org!.id, membership_id: me!.id, request_date: d, shift_code: f.code,
                area_label: off ? null : f.area || null, role_label: off ? null : f.role || null,
                start_time: off ? OFF_START : f.start + ":00", end_time: off ? OFF_END : f.end + ":00", status: "pending"
            }));
            const { data: ins, error } = await supabase.from("shift_requests").insert(rows).select("id");
            if (error) return error.code === "PGRST205" ? "Table shift_requests missing — run backend/sql/002_roster.sql." : error.message;
            const ids = ins!.map(x => x.id);
            pushUndo(async () => { await supabase.from("shift_requests").delete().in("id", ids); });
            say("Request sent — waiting for approval", "success");

        } else if (spec.mode === "assign") {                              // manager -> applied at once
            const err = await overlap(f, dates);
            if (err) return err;
            const { data: ins, error } = await insertShifts(dates.map(d => ({ ...rowOf(f, d), status: "draft", created_by: uid, updated_by: uid })));
            if (error) return error.message;
            const ids = ins!.map(x => x.id);
            pushUndo(async () => { await supabase.from("shifts").delete().in("id", ids); });
            say("Shift assigned", "success");

        } else if (spec.mode === "edit") {
            const old = data.shifts.find(x => x.id === spec.shift.id)!;   // raw row (spec.shift is the manager view)
            const err = await overlap(f, [f.date], old.id);
            if (err) return err;
            const { organization_id: _o, ...patch } = rowOf(f, f.date);
            if (old.status === "draft") {                                  // never published: edit directly
                const e = await updateShift(old.id, { ...patch, updated_by: uid });
                if (e) return e.message;
                pushUndo(async () => {
                    await updateShift(old.id, {
                        shift_date: old.shift_date, membership_id: old.membership_id, start_time: old.start_time, end_time: old.end_time,
                        shift_code: old.shift_code ?? null, area_label: old.area_label ?? null, role_label: old.role_label
                    });
                });
            } else {                                                       // published: pending change until Publish
                const prev = old.draft ?? null;
                const e = await setDraft(old.id, patch);
                if (e) return e;
                pushUndo(async () => { await setDraft(old.id, prev); });
            }
            say("Shift updated — publish to apply", "success");

        } else {                                                          // approve (as is, or edited)
            const req = spec.req;
            const err = await overlap(f, [f.date]);
            if (err) return err;
            const { data: ins, error } = await insertShifts([{ ...rowOf(f, f.date), status: "draft", created_by: uid, updated_by: uid }]);
            if (error) return error.message;
            const sid = ins![0].id;
            await supabase.from("shift_requests").update({ status: "approved", reviewed_by: uid, reviewed_at: new Date().toISOString() }).eq("id", req.id);
            pushUndo(async () => {
                await supabase.from("shifts").delete().eq("id", sid);
                await supabase.from("shift_requests").update({ status: "pending", reviewed_by: null, reviewed_at: null }).eq("id", req.id);
            });
            say("Request approved", "success");
        }

        setSelected(new Set());
        await load();
        return null;
    }

    async function removeOrReject(spec: PopupSpec): Promise<string | null> {
        if (spec.mode === "edit") {
            const old = data.shifts.find(x => x.id === spec.shift.id)!;
            if (old.status === "draft") {                                  // never published: delete for real
                const { error } = await supabase.from("shifts").delete().eq("id", old.id);
                if (error) return error.message;
                pushUndo(async () => { await insertShifts([old as unknown as Record<string, unknown>]); });
                say("Shift deleted", "success");
            } else {                                                       // published: marked until Publish
                const prev = old.draft ?? null;
                const e = await setDraft(old.id, { _delete: true });
                if (e) return e;
                pushUndo(async () => { await setDraft(old.id, prev); });
                say("Shift marked for deletion — publish to apply", "success");
            }
        } else if (spec.mode === "request") {
            const req = spec.req;
            const { error } = await supabase.from("shift_requests").delete().eq("id", req.id);
            if (error) return error.message;
            pushUndo(async () => { await supabase.from("shift_requests").insert(req); });
            say("Request rejected", "success");
        }
        await load();
        return null;
    }

    // ---------- render ----------

    const weekdayName = (i: number) =>   // 2024-01-01 is a Monday
        new Intl.DateTimeFormat(user.language, { weekday: width > 700 ? "long" : "short" }).format(new Date(2024, 0, 1 + i));
    const monthName = new Intl.DateTimeFormat(user.language, { month: "long", year: "numeric" }).format(month);
    const today = startOfToday();
    const sorted = [...selected].sort();

    return (
        <div className="dsh fill">
            <div className="dsh-h split">
                <h2>Roster</h2>
                <button className="ro-export" onClick={() => say("Export: coming soon", "info")}>Export</button>
            </div>

                <div className="ro-bar">

                            </div>

        </div>
    );

/** Shift presets (F, FX, M, U, Sch ...) + areas. Stored in organizations.settings. */
function Presets() {

    const { org, patchOrg } = useAuth();
    const log = useLog();
    const { say } = useStatus();
    const s = org!.settings ?? {};
    const [types, setTypes] = useState<ShiftType[]>(() => shiftTypesOf(s).map(t => ({ ...t })));
    const [areas, setAreas] = useState((s.areas?.length ? s.areas : DEFAULT_AREAS).join("\n"));
    const [busy, setBusy] = useState(false);

    const edit = (i: number, patch: Partial<ShiftType>) => setTypes(t => t.map((x, j) => (j === i ? { ...x, ...patch } : x)));

    async function save() {
        setBusy(true);
        const shift_types = types.filter(t => t.code.trim()).map(t => ({
            code: t.code.trim(), name: t.name.trim() || t.code.trim(), color: t.color, kind: t.kind ?? "work",
            ...((t.kind ?? "work") === "work" ? { start: t.start, end: t.end } : {})
        }));
        const settings = { ...s, shift_types, areas: areas.split("\n").map(x => x.trim()).filter(Boolean) };

        const { error } = await supabase.from("organizations").update({ settings }).eq("id", org!.id);
        setBusy(false);
        if (error) { say(error.message, "error"); return; }

        patchOrg({ settings });
        void log("settings.update", "Shift presets updated", { shift_types: shift_types.length });
        say("Änderungen gespeichert", "success");
    }

    return (
        <>
            <span className="plan-entry-meta">Code · name · color · type · default start/end. “Off” types (e.g. U, Sch) fill the whole day.</span>
            {types.map((t, i) => (
                <div key={i} className="preset-row">
                    <input placeholder="Code" value={t.code} onChange={e => edit(i, { code: e.target.value })} />
                    <input placeholder="Name" value={t.name} onChange={e => edit(i, { name: e.target.value })} />
                    <input type="color" value={t.color} onChange={e => edit(i, { color: e.target.value })} />
                    <select value={t.kind ?? "work"} onChange={e => edit(i, { kind: e.target.value as ShiftType["kind"] })}>
                        <option value="work">Work</option>
                        <option value="off">Off</option>
                    </select>
                    <input type="time" disabled={t.kind === "off"} value={t.start ?? ""} onChange={e => edit(i, { start: e.target.value })} />
                    <input type="time" disabled={t.kind === "off"} value={t.end ?? ""} onChange={e => edit(i, { end: e.target.value })} />
                    <button type="button" onClick={() => setTypes(x => x.filter((_, j) => j !== i))} title="Remove">✕</button>
                </div>
            ))}
            <button type="button" className="preset-add" onClick={() => setTypes(t => [...t, { code: "", name: "", color: "#6b7280", kind: "work", start: "09:00", end: "17:00" }])}>+ Add shift</button>

            <span className="plan-entry-meta" style={{ display: "block", marginTop: 14 }}>Areas — one per line</span>
            <textarea id="setRoles" rows={4} value={areas} onChange={e => setAreas(e.target.value)} />

            <button className="att-btn" disabled={busy} onClick={save}>Save presets</button>
        </>
    );
}
